/**
 * Research-only probability diagnostics. These probabilities approximate nine-inning
 * scoring with independent Poisson runs; they are NOT calibrated MLB win/cover rates.
 * Extra innings, correlated offense, overdispersion, bullpen usage and weather are not
 * accounted for. Values must NEVER change formal model.markets.*.pick by themselves.
 */
function poissonRunProbabilities(expected       )               {
 if(!Number.isFinite(expected)||expected<0.5||expected>20)return null;
 let term=Math.exp(-expected),sum=term;
 const probabilities=[term];
 for(let runs=1;runs<=90;runs++){
  term*=expected/runs;
  probabilities.push(term);
  sum+=term;
  if(runs>=expected&&1-sum<1e-12)break;
 }
 if(!Number.isFinite(sum)||sum<0.99999999)return null;
 return probabilities.map(v=>v/sum);
}

 typeof v==='number'&&Number.isFinite(v)&&v>1&&v<=1000?v:null;
const probabilityRounding=(v       )=>round(v,6);
function projectedProbabilityResearch(projection    ,odds    ,asOf       )          {
 const h=poissonRunProbabilities(projection?.home_runs),a=poissonRunProbabilities(projection?.away_runs);
 if(!h||!a)return null;
 const sourceTime=Date.parse(odds?.observed_at||'');
 const readTime=Date.parse(asOf);
 const trustedMarket=odds?.status==='found'&&typeof odds?.bookmakers_count==='number'&&
  odds.bookmakers_count>0&&Number.isFinite(sourceTime)&&Number.isFinite(readTime)&&sourceTime<=readTime;
 const lines=trustedMarket?odds?.lines||{}:{};
 const totalLine=typeof lines.total==='number'&&Number.isFinite(lines.total)&&lines.total>0&&lines.total<40?lines.total:null;
 const spreadLine=typeof lines.home_spread==='number'&&Number.isFinite(lines.home_spread)&&Math.abs(lines.home_spread)<=30?lines.home_spread:null;
 let homeWin=0,awayWin=0,tie=0,over=0,under=0,totalPush=0,spreadHome=0,spreadAway=0,spreadPush=0;
 for(let i=0;i<h.length;i++)for(let j=0;j<a.length;j++){
  const weight=h[i]*a[j];
  if(i>j)homeWin+=weight;
  else if(i<j)awayWin+=weight;
  else tie+=weight;
  if(totalLine!==null){
   const diff=i+j-totalLine;
   if(diff>1e-9)over+=weight;else if(diff< -1e-9)under+=weight;else totalPush+=weight;
  }
  if(spreadLine!==null){
   const diff=i-j+spreadLine;
   if(diff>1e-9)spreadHome+=weight;else if(diff< -1e-9)spreadAway+=weight;else spreadPush+=weight;
  }
 }
 // Tie probability is reported separately; renormalization is a proxy ONLY,
 // not an actual extra-inning forecast.
 const nonTie=homeWin+awayWin;
 const homeProxy=nonTie>0?homeWin/nonTie:null;
 const awayProxy=nonTie>0?awayWin/nonTie:null;
 const homePrice=trustedMarket?validatedOddsPrice(lines.home_moneyline_decimal):null;
 const awayPrice=trustedMarket?validatedOddsPrice(lines.away_moneyline_decimal):null;
 const fullMLPrice=homePrice!==null&&awayPrice!==null;
 const marketSum=fullMLPrice?1/homePrice +1/awayPrice :null;
 const overPrice=trustedMarket?validatedOddsPrice(lines.over_decimal):null;
 const underPrice=trustedMarket?validatedOddsPrice(lines.under_decimal):null;
 const spreadHomePrice=trustedMarket?validatedOddsPrice(lines.home_spread_decimal):null;
 const spreadAwayPrice=trustedMarket?validatedOddsPrice(lines.away_spread_decimal):null;
 const unitEv=(win       ,loss       ,price            )=>
  price===null?null:probabilityRounding(win*(price-1)-loss);
 const result    ={
  version:'habao-mlb-probability-research-v1',research_only:true,
  affects_official_pick:false,
  model:'independent_poisson_nine_inning_proxy_uncalibrated',
  limitations:['unvalidated_probability_calibration','no_extra_inning_model','no_bullpen_weather_adjustment','no_score_overdispersion'],
  source:{bookmaker_observation_usable:trustedMarket,observed_at:odds?.observed_at||null,
   bookmaker_opening_verified:false},
  score:{expected_home_runs:projection.home_runs,expected_away_runs:projection.away_runs,
   nine_inning_home_outscores:probabilityRounding(homeWin),nine_inning_away_outscores:probabilityRounding(awayWin),
   nine_inning_tie:probabilityRounding(tie)},
  moneyline:{home_win_proxy:homeProxy===null?null:probabilityRounding(homeProxy),
   away_win_proxy:awayProxy===null?null:probabilityRounding(awayProxy),
   market_home_no_vig:marketSum===null?null:probabilityRounding((1/homePrice )/marketSum),
   market_away_no_vig:marketSum===null?null:probabilityRounding((1/awayPrice )/marketSum),
   market_overround:marketSum===null?null:probabilityRounding(marketSum),
   home_ev_proxy:!fullMLPrice||homeProxy===null?null:unitEv(homeProxy,1-homeProxy,homePrice),
   away_ev_proxy:!fullMLPrice||awayProxy===null?null:unitEv(awayProxy,1-awayProxy,awayPrice)},
  totals:{line:totalLine,over_probability:totalLine===null?null:probabilityRounding(over),
   under_probability:totalLine===null?null:probabilityRounding(under),
   push_probability:totalLine===null?null:probabilityRounding(totalPush),
   over_ev_proxy:totalLine===null?null:unitEv(over,under,overPrice),
   under_ev_proxy:totalLine===null?null:unitEv(under,over,underPrice)},
  spread:{home_line:spreadLine,home_cover_probability:spreadLine===null?null:probabilityRounding(spreadHome),
   away_cover_probability:spreadLine===null?null:probabilityRounding(spreadAway),
   push_probability:spreadLine===null?null:probabilityRounding(spreadPush),
   home_ev_proxy:spreadLine===null?null:unitEv(spreadHome,spreadAway,spreadHomePrice),
   away_ev_proxy:spreadLine===null?null:unitEv(spreadAway,spreadHome,spreadAwayPrice)}

}

const mlbClamp=(x,lo,hi)=>Math.max(lo,Math.min(hi,x));
function mlbStatRows(payload,teamId,season,group){
 const blocks=(payload?.stats||[]).filter(x=>String(x.group?.displayName||'').toLowerCase()===group);
 const rows=blocks.flatMap(x=>x.splits||[]).filter(x=>Number(x.team?.id)===Number(teamId)&&Number(x.season)===Number(season)&&Number(x.player?.id)>0);
 // Duplicate player splits are ambiguous; do not double-count or choose arbitrarily.
 const counts=new Map();for(const r of rows)counts.set(r.player.id,(counts.get(r.player.id)||0)+1);
 return rows.filter(r=>counts.get(r.player.id)===1).map(r=>({id:Number(r.player.id),name:r.player.fullName,stat:r.stat||{}}));
}

function mlbSideAdjustment(context,side,stats,team){
 const out={side,mode:'not_applied',batting_fraction:0,bullpen_opponent_runs:0,lineup_applied:false,injury_applied:false,used:[],skipped:[]};
 const lineup=context.lineups?.teams?.find(t=>t.side===side),injuries=(context.availability?.items||[]).filter(p=>p.side===side&&p.status==='out');
 const injured=new Set(injuries.map(p=>Number(p.id)));
 if(stats?.status!=='data_returned'){out.skipped.push('player_stats_unavailable');return out;}
 const batters=stats.hitters.filter(p=>n(p.stat.plateAppearances)>0&&n(p.stat.ops)!==null&&n(p.stat.ops)>=0&&n(p.stat.ops)<=3);
 const totalPa=batters.reduce((s,p)=>s+Number(p.stat.plateAppearances),0);
 const baseline=totalPa?batters.reduce((s,p)=>s+Number(p.stat.plateAppearances)*Number(p.stat.ops),0)/totalPa:0;
 const shrunk=p=>baseline+(Number(p.stat.ops)-baseline)*Number(p.stat.plateAppearances)/(Number(p.stat.plateAppearances)+100);
 const active=new Set((lineup?.active_roster_ids||[]).map(Number));
 if(totalPa>=500&&baseline>0){
  if(lineup?.status==='confirmed'){
   const ids=(lineup.batting_order||[]).map(p=>Number(p.id)),unique=new Set(ids);
   const players=ids.map(id=>batters.find(p=>p.id===id));
   if(ids.length!==9||unique.size!==9||players.some(p=>!p))out.skipped.push('confirmed_lineup_missing_player_stats');
   else if(ids.some(id=>injured.has(id)))out.skipped.push('lineup_injury_conflict');
   else{
    // Equal batting slots: no invented PA advantage for batting-order position.
    const lineupOps=players.reduce((s,p)=>s+shrunk(p),0)/9;
    out.batting_fraction=mlbClamp(lineupOps/baseline-1,-.20,.20);
    out.lineup_applied=true;out.mode='confirmed_lineup';
    out.used=players.map(p=>({id:p.id,plate_appearances:Number(p.stat.plateAppearances),ops:Number(p.stat.ops),shrunk_ops:round(shrunk(p),4)}));
    out.skipped.push('hitter_injuries_already_represented_by_lineup');
   }
  }else if(lineup?.roster_status==='data_returned'){
   const available=batters.filter(p=>active.has(p.id)&&!injured.has(p.id));
   // Backup pool is the lower-PA active hitters, not an invented replacement value.
   const regulars=[...available].sort((a,b)=>Number(b.stat.plateAppearances)-Number(a.stat.plateAppearances)).slice(0,9);
   const regularIds=new Set(regulars.map(p=>p.id));
   const bench=available.filter(p=>!regularIds.has(p.id)),benchPa=bench.reduce((s,p)=>s+Number(p.stat.plateAppearances),0);
   const missing=batters.filter(p=>injured.has(p.id)&&Number(p.stat.plateAppearances)>=30);
   if(missing.length&&benchPa>=30){
    const replacement=bench.reduce((s,p)=>s+shrunk(p)*Number(p.stat.plateAppearances),0)/benchPa;
    out.batting_fraction=mlbClamp(missing.reduce((s,p)=>s+(replacement-shrunk(p))/baseline*Number(p.stat.plateAppearances)/totalPa,0),-.20,.20);
    out.injury_applied=true;out.mode='injury_replacement';
    out.used=missing.map(p=>({id:p.id,plate_appearances:Number(p.stat.plateAppearances),shrunk_ops:round(shrunk(p),4),replacement_ops:round(replacement,4),historical_pa_share:round(Number(p.stat.plateAppearances)/totalPa,4)}));
   }else out.skipped.push(missing.length?'replacement_sample_insufficient':'no_qualified_injured_hitter');
  }else out.skipped.push('roster_unavailable');
 }else out.skipped.push('team_batting_sample_insufficient');
 // Relief injuries affect opponent scoring independently of the batting lineup.
 const relievers=stats.pitchers.filter(p=>innings(p.stat.inningsPitched)>=10&&n(p.stat.era)!==null&&n(p.stat.era)>=0&&Number(p.stat.gamesStarted||0)/Math.max(1,Number(p.stat.gamesPitched||0))<.2);
 const ready=relievers.filter(p=>active.has(p.id)&&!injured.has(p.id));
 const readyIp=ready.reduce((s,p)=>s+innings(p.stat.inningsPitched),0);
 if(lineup?.roster_status==='data_returned'&&readyIp>=20&&Number(team.games_played)>=5&&n(team.era)!==null){
  const era=p=>(Number(p.stat.era)*innings(p.stat.inningsPitched)+Number(team.era)*20)/(innings(p.stat.inningsPitched)+20);
  const replacement=ready.reduce((s,p)=>s+era(p)*innings(p.stat.inningsPitched),0)/readyIp;
  const missing=relievers.filter(p=>injured.has(p.id));
  if(missing.length){
   out.bullpen_opponent_runs=mlbClamp(missing.reduce((s,p)=>s+(replacement-era(p))*Math.min(1,innings(p.stat.inningsPitched)/Number(team.games_played))/9,0),-.35,.35);
   out.injury_applied=true;
   out.bullpen_used=missing.map(p=>({id:p.id,innings:innings(p.stat.inningsPitched),shrunk_era:round(era(p),4),replacement_era:round(replacement,4)}));
  }
 }
 out.batting_fraction=round(out.batting_fraction,6);out.bullpen_opponent_runs=round(out.bullpen_opponent_runs,6);
 return out;
}
function applyMlbPlayers(projection,context,homeStats,awayStats,hs,as){
 const home=mlbSideAdjustment(context,'home',homeStats,hs),away=mlbSideAdjustment(context,'away',awayStats,as);
 // Base projection mixes team attack and opponent defense 50/50. Change attack share only.
 const hd=Number(hs.runs_per_game)*.5*home.batting_fraction+away.bullpen_opponent_runs;
 const ad=Number(as.runs_per_game)*.5*away.batting_fraction+home.bullpen_opponent_runs;
 const h=Math.max(.5,projection.home_runs+hd),a=Math.max(.5,projection.away_runs+ad);
 const adjustment={version:MLB_PLAYER_MODEL,status:[home,away].some(x=>x.lineup_applied||x.injury_applied)?'applied':'not_applied',lineup_adjustment_applied:home.lineup_applied||away.lineup_applied,injury_adjustment_applied:home.injury_applied||away.injury_applied,home,away,home_run_delta:round(h-projection.home_runs,4),away_run_delta:round(a-projection.away_runs,4),baseline_projection:projection,limitations:'OPS run-rate proxy and relief replacement heuristics; not backtest-calibrated. Missing inputs are skipped, not zero-filled. Confirmed batting lineups replace hitter injury adjustments.'};
 return {projection:{...projection,home_runs:round(h),away_runs:round(a),total_runs:round(h+a),home_run_margin:round(h-a),winner:h>=a?'home':'away',method:'season_recent_starters_plus_players_v0.4'},adjustment};
}

function innings(v){const m=/^(\d+)(?:\.([012]))?$/.exec(String(v??''));return m?Number(m[1])+Number(m[2]||0)/3:0;}
function projectMlb(hs,as,hp,ap,hr,ar){
 if(![hs.runs_per_game,hs.runs_allowed_per_game,as.runs_per_game,as.runs_allowed_per_game].every(Number.isFinite)||hs.games_played<5||as.games_played<5)return null;
 const blend=(base,v,count)=>{const w=Number.isFinite(v)?.25*Math.min(10,count||0)/10:0;return base*(1-w)+(Number.isFinite(v)?v:base)*w;};
 let h=(blend(hs.runs_per_game,hr?.runs_per_game,hr?.games_used)+blend(as.runs_allowed_per_game,ar?.runs_allowed_per_game,ar?.games_used))/2;
 let a=(blend(as.runs_per_game,ar?.runs_per_game,ar?.games_used)+blend(hs.runs_allowed_per_game,hr?.runs_allowed_per_game,hr?.games_used))/2;
 const hd=Number.isFinite(hp.era)&&innings(hp.innings)>=20&&Number.isFinite(hs.era)?Math.max(-.65,Math.min(.65,(hp.era-hs.era)*.15)):0;
 const ad=Number.isFinite(ap.era)&&innings(ap.innings)>=20&&Number.isFinite(as.era)?Math.max(-.65,Math.min(.65,(ap.era-as.era)*.15)):0;
 h=Math.max(.5,h+.14+ad);a=Math.max(.5,a+hd);
 return {home_runs:round(h),away_runs:round(a),total_runs:round(h+a),home_run_margin:round(h-a),winner:h>=a?'home':'away',method:'season_recent10_blend_plus_qualified_starters_v0.3',starter_impact:{home_starter_opponent_delta:round(hd,4),away_starter_opponent_delta:round(ad,4),home_qualified:innings(hp.innings)>=20&&Number.isFinite(hp.era),away_qualified:innings(ap.innings)>=20&&Number.isFinite(ap.era)}};
}
async function mlbPlayerContext(gameId,expected,now=new Date().toISOString()){
 const feed=await fetchJson('https://statsapi.mlb.com/api/v1.1/game/'+gameId+'/feed/live');const gd=feed.gameData||{};
 if(Number(gd.game?.pk)!==Number(gameId)||Number(gd.teams?.home?.id)!==Number(expected.home.id)||Number(gd.teams?.away?.id)!==Number(expected.away.id)||Math.abs(Date.parse(gd.datetime?.dateTime)-Date.parse(expected.kickoff_at))>60000)throw Error('mlb_context_fixture_mismatch');
 const teams=[],items=[],pitchers={};const day=new Date(now).toLocaleDateString('en-CA',{timeZone:'America/New_York'});
 for(const side of ['home','away']){
  const t=gd.teams[side],box=feed.liveData?.boxscore?.teams?.[side]||{},order=Array.isArray(box.battingOrder)?box.battingOrder:[];
  const unique=[...new Set(order.map(Number).filter(x=>Number.isSafeInteger(x)&&x>0))],players=box.players||{};
  const names=unique.map(id=>({id,name:players['ID'+id]?.person?.fullName||gd.players?.['ID'+id]?.fullName||''}));
  const confirmed=names.length===9&&names.every(x=>x.name);teams.push({side,team_id:t.id,status:confirmed?'confirmed':names.length?'partial':'not_published',batting_order:names});
  const pitcher=gd.probablePitchers?.[side];pitchers[side]=pitcher?.id?{id:pitcher.id,name:pitcher.fullName,status:'expected'}:null;
  try{const roster=await fetchJson('https://statsapi.mlb.com/api/v1/teams/'+t.id+'/roster?rosterType=fullRoster&date='+day);if(!Array.isArray(roster.roster))throw Error('invalid_roster');
   teams.find(x=>x.side===side).active_roster_ids=roster.roster.filter(p=>p.status?.code==='A').map(p=>Number(p.person?.id));
   for(const p of roster.roster)if(/^D(7|10|15|60)$|^IL/.test(String(p.status?.code||''))||/injured|disabled/i.test(String(p.status?.description||'')))items.push({side,team_id:t.id,id:p.person?.id,name:p.person?.fullName,status:'out',reason:p.status.description});
   teams.find(x=>x.side===side).roster_status='data_returned';
  }catch{teams.find(x=>x.side===side).roster_status='source_unavailable';}
 }
 const rosterOk=teams.every(t=>t.roster_status==='data_returned');
 return {availability:{source:'mlb_official_dated_full_roster',status:rosterOk?(items.length?'data_returned':'no_rows_returned'):'partial',items,fetched_at:now},lineups:{source:'mlb_official_game_feed',status:teams.every(t=>t.status==='confirmed')?'data_returned':teams.some(t=>t.status==='confirmed')?'partial_confirmed':teams.some(t=>t.batting_order.length)?'partial':'not_published',teams,probable_pitchers:pitchers,fetched_at:now},fixture_status:gd.status?.abstractGameState||'',raw_started:gd.status?.abstractGameState==='Live'||gd.status?.abstractGameState==='Final'};
}
async function currentMlbOdds(fixture,now){
 const empty=reason=>({available:false,home_spread:null,total:null,reason,quotes:[],origin:'current_stage_market',fetched_at:now});if(!ODDS_KEY)return empty('odds_key_missing');
 if(!await rpc('reserve_mlb_stage_api',{p_provider:'odds',p_cost:1,p_limit:Math.max(1,Math.min(300,Number(Deno.env.get('MLB_STAGE_DAILY_ODDS_LIMIT')||24)))}))return empty('odds_daily_budget');
 const start=new Date(Date.parse(fixture.kickoff_at)-15*60000).toISOString().replace('.000Z','Z'),end=new Date(Date.parse(fixture.kickoff_at)+15*60000).toISOString().replace('.000Z','Z');
 const p=await fetchJson('https://api.the-odds-api.com/v4/sports/baseball_mlb/odds?'+new URLSearchParams({apiKey:ODDS_KEY,regions:'us',markets:'spreads,totals',oddsFormat:'decimal',commenceTimeFrom:start,commenceTimeTo:end}));
 if(!Array.isArray(p))return empty('odds_invalid');const event=p.find(x=>x.home_team===fixture.home.name&&x.away_team===fixture.away.name&&Math.abs(Date.parse(x.commence_time)-Date.parse(fixture.kickoff_at))<=60000);if(!event)return empty('odds_event_missing');
 const quotes=[];for(const book of event.bookmakers||[]){const spreads=book.markets?.find(x=>x.key==='spreads'),totals=book.markets?.find(x=>x.key==='totals'),h=spreads?.outcomes?.find(x=>x.name===fixture.home.name),a=spreads?.outcomes?.find(x=>x.name===fixture.away.name),over=totals?.outcomes?.find(x=>x.name==='Over'),under=totals?.outcomes?.find(x=>x.name==='Under');
  if((spreads&&Date.parse(spreads.last_update||book.last_update)>Date.parse(now))||(totals&&Date.parse(totals.last_update||book.last_update)>Date.parse(now)))continue;
  const spread=Number.isFinite(h?.point)&&Number.isFinite(a?.point)&&h.point===-a.point?h.point:null,total=Number.isFinite(over?.point)&&over.point===under?.point?over.point:null;if(spread===null&&total===null)continue;
  quotes.push({bookmaker:book.key,home_spread:spread,away_spread:spread===null?null:-spread,total,over_odds:over?.price??null,under_odds:under?.price??null});}
 const q=quotes.find(x=>x.home_spread!==null&&x.total!==null)||quotes[0];return q?{...q,quotes,available:true,event_id:event.id,origin:'current_stage_market',fetched_at:now}:empty('odds_markets_missing');
}
